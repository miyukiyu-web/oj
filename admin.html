📜 Google Apps Script 後端程式碼（自動建表 + 班級讀寫）

        
      

      

        
建立空白 Google 試算表 → 擴充功能 → Apps Script → 貼上以下程式碼並部署為網頁應用程式（誰可以存取選「所有人」）：

        
const SHEET_NAME = "Students";
const HEADERS = [
  "班級", "帳號", "密碼", "姓名",
  "單元一", "單元二", "單元三", "單元四", "單元五", "單元六",
  "總題數", "已通過題號JSON", "程式碼備份JSON", "最後更新時間"
];

function getOrInitSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
    sheet.setFrozenRows(1);
    sheet.getRange("A:D").setNumberFormat("@");
  }
  return sheet;
}

function doGet(e) {
  const sheet = getOrInitSheet();
  const rows = sheet.getDataRange().getValues();
  const students = [];
  for (let i = 1; i < rows.length; i++) {
    if (!rows[i][1] || !rows[i][3]) continue;
    students.push({
      className: String(rows[i][0] || "未分班").trim(),
      account: String(rows[i][1]).trim(),
      password: String(rows[i][2] || "").trim(),
      name: String(rows[i][3]).trim(),
      scores: [
        Number(rows[i][4] || 0), Number(rows[i][5] || 0), Number(rows[i][6] || 0),
        Number(rows[i][7] || 0), Number(rows[i][8] || 0), Number(rows[i][9] || 0)
      ],
      solvedIds: rows[i][11] ? JSON.parse(rows[i][11]) : [],
      savedCodes: rows[i][12] ? JSON.parse(rows[i][12]) : {},
      updatedAt: rows[i][13] ? String(rows[i][13]) : ""
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", students }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const data = JSON.parse(e.postData.contents);
    const sheet = getOrInitSheet();
    const rows = sheet.getDataRange().getValues();
    let targetRow = -1;
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][1]).trim().toLowerCase() === String(data.account).trim().toLowerCase()) {
        targetRow = i + 1;
        break;
      }
    }
    if (targetRow === -1) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "找不到該帳號" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    const s = data.unitScores || [0,0,0,0,0,0];
    const total = s.reduce((a, b) => a + b, 0);
    const nowStr = Utilities.formatDate(new Date(), "GMT+8", "yyyy-MM-dd HH:mm:ss");
    sheet.getRange(targetRow, 5, 1, 10).setValues([[
      s[0], s[1], s[2], s[3], s[4], s[5], total,
      JSON.stringify(data.solvedIds || []),
      JSON.stringify(data.savedCodes || {}),
      nowStr
    ]]);
    return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
